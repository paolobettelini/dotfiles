cmake_minimum_required(VERSION 3.25)
project(hypr-blackhole LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HYPRLAND REQUIRED IMPORTED_TARGET hyprland)
pkg_check_modules(GLES REQUIRED IMPORTED_TARGET glesv2)
add_library(hypr-blackhole SHARED src/plugin.cpp)
set_target_properties(hypr-blackhole PROPERTIES PREFIX "" OUTPUT_NAME "hypr-blackhole")
target_link_libraries(hypr-blackhole PRIVATE PkgConfig::HYPRLAND PkgConfig::GLES)
target_compile_definitions(hypr-blackhole PRIVATE BLACKHOLE_SHADER_PATH="${CMAKE_CURRENT_SOURCE_DIR}/shaders/blackhole.frag")
target_compile_options(hypr-blackhole PRIVATE -Wall -Wextra -Wpedantic -fno-gnu-unique)
